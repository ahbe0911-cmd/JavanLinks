name: Build Windows EXE

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read

concurrency:
  group: javan-windows-build
  cancel-in-progress: true

jobs:
  build:
    runs-on: windows-2022
    timeout-minutes: 35
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
          architecture: x64
      - name: Build and verify executable
        shell: pwsh
        run: ./build_windows.ps1 -Python python
      - name: Upload standalone Windows executable
        uses: actions/upload-artifact@v4
        with:
          name: JavanLinks-Windows-x64
          path: |
            dist/JavanLinks.exe
            dist/JavanLinks.exe.sha256
          if-no-files-found: error
          compression-level: 0
          retention-days: 30
  publish:
    needs: build
    if: github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    permissions:
      contents: write
    steps:
      - uses: actions/download-artifact@v4
        with:
          name: JavanLinks-Windows-x64
          path: release
      - name: Publish verified EXE
        env:
          GH_TOKEN: ${{ github.token }}
          GH_REPO: ${{ github.repository }}
        shell: bash
        run: |
          set -euo pipefail
          cd release
          sha256sum --check JavanLinks.exe.sha256
          if gh release view windows-latest >/dev/null 2>&1; then
            gh release upload windows-latest JavanLinks.exe JavanLinks.exe.sha256 --clobber
          else
            gh release create windows-latest JavanLinks.exe JavanLinks.exe.sha256 --target "$GITHUB_SHA" --title "JavanLinks Windows x64" --notes "Standalone executable; unsigned; Windows 10/11 x64. Build and startup smoke test passed."
          fi

# لینک‌های کاربردی کافینت جوان — فاز ۱

نسخه 0.1.0، اسکلت مستقل نرم‌افزار ویندوزی بر پایه Python و PySide6/Qt WebEngine.

## اجرای ویندوز

پیش‌نیاز پیشنهادی: Windows 10/11 نسخه ۶۴ بیتی، Python 3.11 یا 3.12 نسخه ۶۴ بیتی و اینترنت برای نصب وابستگی‌ها. فایل ZIP را استخراج و `run.bat` را اجرا کنید. یا در PowerShell از پوشه پروژه:

```powershell
py -3.12 -m venv .venv
.\.venv\Scripts\python.exe -m pip install -r requirements.txt
.\.venv\Scripts\python.exe main.py
```

اگر فونت Vazirmatn را در Windows نصب کنید، برنامه از آن استفاده می‌کند. همچنین می‌توانید فایل TTF رسمی را به `assets/fonts/` اضافه کنید. در غیاب فونت وزیرمتن، فونت سیستمی جایگزین می‌شود. فایل فونت همراه این بسته نیست.

## امکانات فاز اول

- پنجره تمام‌اندازه با چیدمان RTL و منوی راست، قابل جمع‌کردن.
- صفحه اصلی داخلی مستقل از اینترنت و شش لینک نمونه در چهار دسته.
- جستجوی زنده نام، دسته، توضیحات و دامنه با نرمال‌سازی حروف فارسی/عربی.
- یک نمای Qt WebEngine، نشانگر پیشرفت و وضعیت خطای بارگذاری.
- پروفایل موقت مرورگر؛ نشست کاربران تا زمان خروج کامل از برنامه برقرار است. این نسخه جایگزین سامانه مدیریت نشست مشتریان کافی‌نت نیست.

## آزمون

```powershell
.\.venv\Scripts\python.exe -m unittest discover -s tests -v
.\.venv\Scripts\python.exe -m tests.smoke_gui
```

آزمون اول بدون Qt نیز اجرا می‌شود؛ آزمون دوم به Qt و محیط گرافیکی نیاز دارد. برای تست دستی، لینک‌های نمونه را باز کنید و نمایش وب‌سایت، ورود، ریدایرکت و خطاهای احتمالی را بررسی کنید. لینک‌ها نمونه‌اند و در این محیط به‌صورت زنده آزموده نشده‌اند.

## مرزبندی فازها

فاز ۲: CRUD، فایل HTML محلی، JSON، پشتیبان‌گیری و بازیابی. فاز ۳: تب‌ها، کنترل مرورگر، چاپ، زوم، دانلود و مدیریت پنجره‌های جدید. فاز ۴: بسته‌بندی EXE، آیکون و GitHub Actions. در فاز ۱ دانلود و پنجره‌های جدید عمداً فعال نشده‌اند. سازگاری تمام سامانه‌های بانکی/دولتی یا کاهش مصرف حافظه در مقایسه با Chrome تضمین نشده است؛ Qt WebEngine نیز مبتنی بر Chromium است. از مرورگر روی سامانه‌های حساس تا تکمیل و آزمون امنیت و سازگاری استفاده عملیاتی نشود.

## ساختار

`main.py`: نقطه ورود؛ `javan/app.py`: راه‌اندازی؛ `catalog.py`: مدل و جستجو؛ `sidebar.py`: منوی خدمات؛ `browser.py`: مرورگر؛ `window.py`: پنجره؛ `theme.py`: ظاهر و فونت؛ `tests/`: آزمون‌ها.


## فاز ۲ — نسخه 0.2.0

- دکمه مدیریت لینک‌ها: افزودن، ویرایش، حذف، دسته‌بندی قابل تایپ، انتخاب HTML و جستجوی زنده.
- ذخیره خودکار در `catalog.json` در پوشه داده محلی Qt (معمولاً `%LOCALAPPDATA%\Javan\Javan Links` در ویندوز). نوشتن اتمیک؛ در خرابی فایل اصلی برنامه داده را بازنویسی نمی‌کند.
- پشتیبان JSON شامل لینک‌ها و وضعیت جمع‌شدن منو است. بازیابی کل فهرست و تنظیمات را جایگزین می‌کند و قبل از آن `catalog.previous.json` را می‌نویسد. فایل‌های HTML، دارایی‌های مرتبط، نشست وب و داده داخلی HTML در JSON نیستند. مسیرهای محلی مطلق‌اند و روی سیستم مقصد ممکن است به ویرایش نیاز داشته باشند.
- HTML از فایل اصلی باز می‌شود و حذف لینک فایل را حذف نمی‌کند. ابزار محلی پروفایل جدا دارد و درخواست‌های شبکه و فایل خارج از پوشه ابزار محدود می‌شوند؛ فقط فایل‌های مورد اعتماد اجرا شوند. داده محلی WebEngine در پوشه جداگانه نگه داشته می‌شود؛ رفتار localStorage در file:// باید روی ویندوز و هر ابزار آزمایش شود.
- تب، چاپ، دانلود و ساخت EXE هنوز در این فاز فعال نیستند. بخش‌های فاز ۱ در بالا سابقه نسخه قبلی هستند.
- تست منطق: `python -m unittest discover -s tests -v`. تست گرافیکی: `python -m tests.smoke_gui`. برای تست دستی فاز دوم، یک HTML ساده و یک وب‌سایت اضافه کنید، برنامه را بسته و باز کنید، یک لینک را ویرایش یا حذف کنید، پشتیبان بگیرید و بازیابی کنید؛ مسیر HTML مفقود را نیز آزمایش کنید.

- پنجره جدید ناشی از تعامل کاربر در تب داخلی باز می‌شود؛ پنجره‌های خودکار مسدود می‌شوند و ممکن است برخی ورودهای بانکی نیازمند بررسی باشند.
- ابزارهای HTML از وب جدا و به پروفایل پایدار اختصاصی هر پوشه منتقل شده‌اند. فایل‌های اصلی و JSON قبلی دست‌نخورده‌اند؛ داده داخلی مرورگر محلی فاز دوم خودکار به پروفایل‌های جدید منتقل نمی‌شود و پوشه‌های قدیمی حفظ می‌شوند. برای ابزارهای دارای داده مهم، پیش از ارتقا از داخل خود ابزار پشتیبان بگیرید.
- میانبرها: Ctrl+T / Ctrl+W / Ctrl+Tab / Ctrl+L / Ctrl+R / Ctrl+P / Ctrl+J / Ctrl+= / Ctrl+- / Ctrl+0 و Alt+Left/Right.
- آزمون منطق: python -m unittest discover -s tests -v ؛ آزمون GUI: python -m tests.smoke_gui. چاپ واقعی، دانلود، بازشدن پنجره ورود، PDF، ابزار محلی دارای localStorage و چند تب را روی ویندوز آزمایش کنید.
- فاز ۴ (EXE و Workflow) هنوز شروع نشده است.


## فاز ۴ — نسخه 0.4.0

پس از موفقیت ساخت GitHub Actions، فایل مستقل `JavanLinks.exe` و SHA256 از صفحه Releases یا Artifact به نام JavanLinks-Windows-x64 قابل دریافت است.

- Windows 10/11 x64؛ نصب Python روی سیستم مقصد لازم نیست. Qt WebEngine در بسته تک‌فایلی قرار دارد و هنگام شروع در پوشه موقت استخراج می‌شود؛ شروع نخست ممکن است کند باشد.
- آیکون اختصاصی با منبع SVG و اسکریپت ساخت ICO چنداندازه. فونت Vazirmatn از Google Fonts به همراه متن مجوز SIL OFL در مرحله ساخت دریافت و در برنامه بسته‌بندی می‌شود.
- EXE امضای Authenticode ندارد؛ هش SHA256 برای کنترل یکپارچگی ارائه می‌شود. استفاده سازمانی و بانکی نیازمند بررسی امنیتی، به‌روزرسانی Qt و تست سامانه‌هاست.
- فایل داده در پوشه محلی ویندوز باقی می‌ماند. پیش از ارتقا از JSON و داده‌های داخل ابزارهای HTML جداگانه پشتیبان بگیرید.
- ساخت محلی با Python 3.12 x64: در PowerShell، `powershell -ExecutionPolicy Bypass -File .\build_windows.ps1`. خروجی `dist\JavanLinks.exe`.
- Workflow پس از هر push به main یا اجرای دستی، آزمون‌های منطق و تست ابتدایی اجرای EXE را انجام می‌دهد؛ فقط پس از موفقیت، خروجی منتشر می‌شود.
- آزمون ابتدایی شامل رندر صفحه اصلی، جستجو، تب‌ها و زوم است. چاپ واقعی، دانلود و ورود سامانه‌ها باید دستی آزمایش شوند.
- مجوز فونت در بسته assets/fonts/OFL.txt قرار دارد. Qt/PySide6 و سایر وابستگی‌ها تحت مجوزهای خودشان هستند؛ پیش از توزیع تجاری، الزامات مجوزهای Qt/PySide6، PyInstaller و Chromium را بررسی کنید.


## نسخه 0.5.0 — حسابداری یکپارچه

دکمه سبز «حسابداری» در هدر، فایل حسابداری v3 را بدون اینترنت در یک تب ثابت باز می‌کند. این تب فقط هنگام اولین کلیک ساخته می‌شود و کلیک مجدد به تب موجود برمی‌گردد. منطق مالی و کلیدهای LocalStorage فایل اصلی حفظ شده‌اند. HTML به مسیر دائمی `accounting/index.html` در پوشه AppLocalDataLocation کپی می‌شود؛ پروفایل اختصاصی پایدار این پوشه از نشست اینترنتی جداست. تغییر فایل EXE یا مسیر استخراج موقت آن، این مسیر و پروفایل را تغییر نمی‌دهد. داده‌های مرورگر قدیمی خارج از این برنامه خودکار وارد نمی‌شوند؛ از قابلیت پشتیبان JSON داخل حسابداری استفاده کنید.

فونت وزیرمتن همراه بسته ساخته می‌شود. دکمه «فونت‌ها» اجازه می‌دهد ZIP/TTFهای روزنامه، نازنین و چهار وزن وزیرمتن را مستقیماً از دیسک وارد کنید؛ فایل‌های اضافی ZIP استخراج نمی‌شوند. روزنامه برای عنوان برند و نازنین برای متن توضیحی فاکتور چاپی A4/A5 استفاده می‌شوند، و وزیرمتن برای رابط و مبالغ. فایل‌های روزنامه و نازنین در مخزن عمومی منتشر نشده‌اند. دسترسی خارجی HTML حذف شده، ساعت در تب پنهان متوقف می‌شود و به‌روزرسانی پیشرفت دانلود به‌صورت تجمیعی انجام می‌شود.

برای حفظ اطلاعات مالی، پشتیبان JSON داخلی حسابداری را به‌طور منظم در مسیر دیگری نگه دارید؛ پشتیبان مدیریت لینک‌ها شامل داده حسابداری نیست. حذف پوشه داده ویندوز یا پروفایل اختصاصی ممکن است اطلاعات را از بین ببرد. تغییرات CPU/RAM باید روی ویندوز واقعی اندازه‌گیری شوند.

      /* -----------------------------------------------------------------------
         ۳. توابع کمکی: تاریخ شمسی، ساعت زنده ایران، تبدیل عدد به حروف فارسی
         ----------------------------------------------------------------------- */
      function gregorianToJalali(gy, gm, gd) {
        const g_d_m = [0, 31, 59, 90, 120, 151, 181, 212, 243, 273, 304, 334];
        const gy2 = gm > 2 ? gy + 1 : gy;
        let days = 355666 + (365 * gy) + Math.floor((gy2 + 3) / 4) - Math.floor((gy2 + 99) / 100) + Math.floor((gy2 + 399) / 400) + gd + g_d_m[gm - 1];
        let jy = -1595 + (33 * Math.floor(days / 12053));
        days %= 12053;
        jy += 4 * Math.floor(days / 1461);
        days %= 1461;
        if (days > 365) {
          jy += Math.floor((days - 1) / 365);
          days = (days - 1) % 365;
        }
        const jm = days < 186 ? 1 + Math.floor(days / 31) : 7 + Math.floor((days - 186) / 30);
        const jd = 1 + (days < 186 ? days % 31 : (days - 186) % 30);
        return { jy, jm, jd };
      }

      function getTodayJalali() {
        const now = new Date();
        const { jy, jm, jd } = gregorianToJalali(now.getFullYear(), now.getMonth() + 1, now.getDate());
        const pad = (n) => String(n).padStart(2, '0');
        return `${jy}/${pad(jm)}/${pad(jd)}`;
      }

        const now = new Date();

        const months = ['فروردین', 'اردیبهشت', 'خرداد', 'تیر', 'مرداد', 'شهریور', 'مهر', 'آبان', 'آذر', 'دی', 'بهمن', 'اسفند'];
        const { jy, jm, jd } = gregorianToJalali(now.getFullYear(), now.getMonth() + 1, now.getDate());
        return `${days[now.getDay()]}، ${jd} ${months[jm - 1]} ${jy}`;
      }

      /**
       * تبدیل دقیق عدد صحیح به حروف فارسی (تا تریلیون)
       */
      function numberToPersianWords(input) {
        const num = Math.floor(Math.abs(Number(input) || 0));
        if (num === 0) return 'صفر';

        const ones = ['', 'یک', 'دو', 'سه', 'چهار', 'پنج', 'شش', 'هفت', 'هشت', 'نه'];
        const teens = ['ده', 'یازده', 'دوازده', 'سیزده', 'چهارده', 'پانزده', 'شانزده', 'هفده', 'هجده', 'نوزده'];
        const tens = ['', '', 'بیست', 'سی', 'چهل', 'پنجاه', 'شصت', 'هفتاد', 'هشتاد', 'نود'];
        const hundreds = ['', 'صد', 'دویست', 'سیصد', 'چهارصد', 'پانصد', 'ششصد', 'هفتصد', 'هشتصد', 'نهصد'];
        const scales = ['', 'هزار', 'میلیون', 'میلیارد', 'تریلیون'];

        function threeDigitsToWords(n) {
          const parts = [];
          const h = Math.floor(n / 100);
          const rem = n % 100;
          if (h > 0) parts.push(hundreds[h]);
          if (rem >= 10 && rem <= 19) {
            parts.push(teens[rem - 10]);
          } else {
            const t = Math.floor(rem / 10);
            const o = rem % 10;
            if (t > 0) parts.push(tens[t]);
            if (o > 0) parts.push(ones[o]);
          }
          return parts.join(' و ');
        }

        const chunks = [];
        let temp = num;
        let scaleIndex = 0;

        while (temp > 0 && scaleIndex < scales.length) {
          const chunk = temp % 1000;
          if (chunk > 0) {
            const words = threeDigitsToWords(chunk);
            const scale = scales[scaleIndex];
            chunks.unshift(scale ? `${words} ${scale}` : words);
          }
          temp = Math.floor(temp / 1000);
          scaleIndex++;
        }

        return chunks.join(' و ');
      }

      /* اعداد فارسی/عربی -> انگلیسی */
      function toEnDigits(str) {
        return String(str ?? '')
          .replace(/[۰-۹]/g, d => String(d.charCodeAt(0) - 1776))
          .replace(/[٠-٩]/g, d => String(d.charCodeAt(0) - 1632));
      }

      /* یکسان‌سازی تاریخ شمسی به قالب YYYY/MM/DD (در صورت نامعتبر بودن، همان ورودی برمی‌گردد) */
      function normalizeJalaliDate(str) {
        const raw = toEnDigits(str).trim();
        const m = raw.match(/^(\d{4})\s*[\/\-.]\s*(\d{1,2})\s*[\/\-.]\s*(\d{1,2})$/);
        if (!m) return raw;
        const mo = Number(m[2]), da = Number(m[3]);
        if (mo < 1 || mo > 12 || da < 1 || da > 31) return raw;
        return `${m[1]}/${String(mo).padStart(2, '0')}/${String(da).padStart(2, '0')}`;
      }

      function isValidJalaliDate(str) {
        return /^\d{4}\/\d{2}\/\d{2}$/.test(str);
      }

      /* بازمحاسبه مبالغ فاکتور از روی اقلام تا تاریخچه، چاپ و CSV همیشه با هم یکسان باشند */
      function normalizeInvoice(inv) {
        if (!inv || typeof inv !== 'object') return null;
        inv.items = Array.isArray(inv.items) ? inv.items : [];
        inv.date = normalizeJalaliDate(inv.date || '');
        const t = computeTotals(inv.items, inv.discountType || 'amount', inv.discountValue || 0);
        inv.subtotal = t.subtotal;
        inv.discountAmount = t.discountAmount;
        inv.finalToman = t.finalToman;
        inv.finalRial = t.finalRial;
        if (!inv.id) inv.id = generateId('inv');
        return inv;
      }

      function debounce(fn, ms = 120) {
        let timer;
        return function (...args) {
          clearTimeout(timer);
          timer = setTimeout(() => fn.apply(this, args), ms);
        };
      }

      function formatNumber(n) {
        return Math.round(Number(n) || 0).toLocaleString('en-US');
      }

      function escapeHtml(str) {
        return String(str ?? '')
          .replace(/&/g, '&amp;')
          .replace(/</g, '&lt;')
          .replace(/>/g, '&gt;')
          .replace(/"/g, '&quot;')
          .replace(/'/g, '&#39;');
      }

      function loadStorage() {
        try {
          const rawServices = localStorage.getItem(KEYS.SERVICES);
          if (rawServices) {
            const parsed = JSON.parse(rawServices);
            state.services = Array.isArray(parsed) && parsed.length > 0 ? parsed : [...DEFAULT_SERVICES];
          } else {
            state.services = [...DEFAULT_SERVICES];
            saveServices();
          }

          const rawInvoices = localStorage.getItem(KEYS.INVOICES);
          if (rawInvoices) {
            const parsedInv = JSON.parse(rawInvoices);
            state.invoices = Array.isArray(parsedInv) ? parsedInv.map(normalizeInvoice).filter(Boolean) : [];
          }

          const rawDebts = localStorage.getItem(KEYS.DEBTS);
          if (rawDebts) {
            const pd = JSON.parse(rawDebts);
            state.debts = Array.isArray(pd) ? pd.filter(d => d && d.id) : [];
          }

          const rawNext = localStorage.getItem(KEYS.NEXT_NUM);
          if (rawNext && !isNaN(Number(rawNext))) {
            state.nextInvoiceNumber = Number(rawNext);
          }

          const rawLogo = localStorage.getItem(KEYS.LOGO);
          if (rawLogo) state.customLogo = rawLogo;

          const rawSettings = localStorage.getItem(KEYS.SETTINGS);
          if (rawSettings) {
            state.settings = { ...DEFAULT_SETTINGS, ...JSON.parse(rawSettings) };
          }

          const savedTheme = localStorage.getItem(KEYS.THEME);
          if (savedTheme === 'dark' || savedTheme === 'light') {
            state.theme = savedTheme;
          }
        } catch (err) {
          console.error('Storage read error:', err);
          state.services = [...DEFAULT_SERVICES];
        }
      }

      function bindDebtEvents() {
        const nameEl = document.getElementById('debt-name');
        const amountEl = document.getElementById('debt-amount');
        const hintEl = document.getElementById('debt-amount-hint');

        document.getElementById('btn-open-debts').addEventListener('click', () => {
          state.debtTab = 'open';
          document.getElementById('debt-search').value = '';
          renderDebtsModal();

          hintEl.textContent = n ? `${formatNumber(n)} تومان — ${numberToPersianWords(n)} تومان` : '';
        });

          if (!name) { showToast('نام خانوادگی را وارد کنید.', 'error'); nameEl.focus(); return; }

            id: generateId('debt'),
            name,
            amount,
            date: getTodayJalali(),
            paid: false,
            paidDate: null,
            createdAt: Date.now()
          });
          saveDebts();
          nameEl.value = '';
          amountEl.value = '';
          hintEl.textContent = '';
          state.debtTab = 'open';

          nameEl.focus();
          showToast(`نسیه «${name}» به مبلغ ${formatNumber(amount)} تومان ثبت شد.`, 'success');
        });

        document.querySelectorAll('#debts-modal [data-debt-tab]').forEach(b => {
          b.addEventListener('click', () => { state.debtTab = b.dataset.debtTab; renderDebtsModal(); });
        });
        document.getElementById('debt-search').addEventListener('input', debounce(renderDebtsModal));

        document.getElementById('debt-tbody').addEventListener('click', (e) => {
          const btn = e.target.closest('[data-debt-act]');
          if (!btn) return;

            d.paidDate = getTodayJalali();

            renderDebtsModal();
            showToast(`نسیه «${d.name}» پرداخت شد ✓`, 'success');

            d.paid = false;
            d.paidDate = null;
            saveDebts();
            renderDebtsModal();
            showToast(`نسیه «${d.name}» دوباره باز شد.`, 'info');

            openConfirm('حذف نسیه', `نسیه «${d.name}» (${formatNumber(d.amount)} تومان) حذف شود؟`, () => {
              state.debts = state.debts.filter(x => x.id !== id);
              saveDebts();
              renderDebtsModal();
              showToast('نسیه حذف شد.', 'info');
            }, 'حذف');
          }
        });
      }

      /* -----------------------------------------------------------------------
         ۶. محاسبات مالی دقیق (تومان، ریال، حروف فارسی)
         ----------------------------------------------------------------------- */
      function computeTotals(items, discountType, discountValue) {
        let subtotal = 0;
        let totalQty = 0;

        for (let i = 0; i < items.length; i++) {
          const q = Math.max(0, Number(items[i].qty) || 0);
          const p = Math.max(0, Number(items[i].price) || 0);
          subtotal += q * p;
          totalQty += q;
        }

        const rawDisc = Math.max(0, Number(discountValue) || 0);
        let discountAmount = 0;
        if (discountType === 'percent') {
          discountAmount = Math.round((subtotal * Math.min(100, rawDisc)) / 100);
        } else {
          discountAmount = Math.min(subtotal, Math.round(rawDisc));
        }

        const finalToman = Math.max(0, subtotal - discountAmount);
        const finalRial = finalToman * 10;
        const wordsToman = `${numberToPersianWords(finalToman)} تومان`;

        return { subtotal, totalQty, discountAmount, finalToman, finalRial, wordsToman };
      }

      /* به‌روزرسانی سریع یک ردیف بدون ساخت مجدد کل جدول */
      function updateRowDOM(idx) {
        const row = document.querySelector(`#invoice-tbody tr[data-idx="${idx}"]`);
        const item = state.invoice.items[idx];
        if (!row || !item) return renderInvoiceTable();
        const qtyInput = row.querySelector('[data-inv-edit="qty"]');
        if (qtyInput) qtyInput.value = item.qty;
        const cell = row.querySelector('.row-total-cell');
        if (cell) cell.textContent = formatNumber((Number(item.qty) || 0) * (Number(item.price) || 0));
        updateSummaryDOM();
      }

        document.getElementById('summary-final-rial').textContent = `${formatNumber(t.finalRial)} ریال`;
        document.getElementById('summary-final-words').textContent = t.wordsToman;
      }

          const catMatch = state.activeCategory === 'همه' || (srv.category || 'سایر') === state.activeCategory;
          const textMatch = !q ||
            srv.name.toLowerCase().includes(q) ||
            (srv.category && srv.category.toLowerCase().includes(q)) ||

      function addSelectedServicesToInvoice() {
        if (state.modalSelectedIds.size === 0) {
          showToast('لطفاً حداقل یک خدمت را تیک بزنید.', 'error');
          return;
        }

        let added = 0;
        let updated = 0;

        state.modalSelectedIds.forEach(id => {
          const srv = state.services.find(s => s.id === id);
          if (!srv) return;
          const exist = state.invoice.items.find(i => i.serviceId === srv.id);
          if (exist) {
            exist.qty = (Number(exist.qty) || 1) + 1;
            updated++;
          } else {
            state.invoice.items.push({
              serviceId: srv.id,
              name: srv.name,
              price: Number(srv.price) || 0,
              qty: 1
            });
            added++;
          }
        });

        renderInvoiceTable();
        closeModal('services-modal');

        if (added > 0 && updated > 0) showToast(`${added} خدمت اضافه و تعداد ${updated} مورد افزایش یافت.`, 'success');
        else if (added > 0) showToast(`${added} خدمت به فاکتور اضافه شد.`, 'success');
        else if (updated > 0) showToast(`تعداد ${updated} خدمت در فاکتور افزایش یافت.`, 'success');
      }

         ۹. مودال تاریخچه و آمار سریع
         ----------------------------------------------------------------------- */
      function renderHistoryModal() {
        const today = getTodayJalali();
        let todaySales = 0;
        let todayCount = 0;
        let totalSales = 0;
        let creditTotal = 0;

        for (let i = 0; i < state.invoices.length; i++) {
          const inv = state.invoices[i];
          const amount = Number(inv.finalToman) || 0;
          const isCredit = inv.paymentStatus === 'نسیه';
          totalSales += amount;
          if (isCredit) creditTotal += amount;
          if (inv.date === today) {
            todayCount++;
            if (!isCredit) todaySales += amount;
          }
        }

        document.getElementById('stat-today-sales').textContent = `${formatNumber(todaySales)} تومان`;
        document.getElementById('stat-today-count').textContent = `${formatNumber(todayCount)} فاکتور`;
        document.getElementById('stat-total-sales').textContent = `${formatNumber(totalSales)} تومان`;
        document.getElementById('stat-credit-total').textContent = `${formatNumber(creditTotal)} تومان`;

        const q = document.getElementById('history-search').value.trim().toLowerCase();
        const filtered = state.invoices.filter(inv =>
          !q ||
          String(inv.invoiceNumber || '').toLowerCase().includes(q) ||
          String(inv.customerName || '').toLowerCase().includes(q) ||
          String(inv.customerPhone || '').toLowerCase().includes(q)
        );

        const tbody = document.getElementById('history-tbody');
        if (filtered.length === 0) {
          tbody.innerHTML = `<tr><td colspan="6"><div class="empty-box" style="padding:26px;">فاکتوری در تاریخچه یافت نشد.</div></td></tr>`;
          return;
        }

        const HISTORY_LIMIT = 150;
        const shown = filtered.slice(0, HISTORY_LIMIT);
        tbody.innerHTML = shown.map(inv => {
          const status = inv.paymentStatus || 'پرداخت شده';
          const badgeClass = status === 'نسیه' ? 'badge-credit' : 'badge-paid';
          return `
            <tr>
              <td class="num-tabular" style="font-weight:800;">#${escapeHtml(inv.invoiceNumber)}</td>
              <td>
                <div style="font-weight:700;">${escapeHtml(inv.customerName || 'مشتری حضوری')}</div>

              <td class="num-tabular" style="color:var(--text-secondary);">${escapeHtml(inv.date)}</td>
              <td><span class="badge ${badgeClass}">${escapeHtml(status)}</span></td>
              <td class="num-tabular" style="font-weight:800; color:var(--emerald-600);">${formatNumber(inv.finalToman)} ت</td>
              <td style="text-align:center;">
                <div style="display:inline-flex; gap:5px;">

                </div>
              </td>

        }).join('') + (filtered.length > HISTORY_LIMIT
          ? `<tr><td colspan="6"><div class="empty-box" style="padding:12px;">${formatNumber(HISTORY_LIMIT)} مورد از ${formatNumber(filtered.length)} نمایش داده شد؛ برای یافتن بقیه جستجو کنید.</div></td></tr>`

      function executePrint(inv) {
        if (!inv.items || inv.items.length === 0) {
          showToast('فاکتور خالی است! ابتدا از بخش «خدمات» اقلام را اضافه کنید.', 'error');
          return;
        }
        const sizeMode = document.getElementById('print-size-select').value;
        const pageStyleEl = document.getElementById('dynamic-print-page-style');

        if (sizeMode === '80mm') {
          pageStyleEl.textContent = '@page { size: 80mm auto; margin: 3mm; }';
        } else if (sizeMode === 'a5') {
          pageStyleEl.textContent = '@page { size: A5 portrait; margin: 8mm; }';
        } else {
          pageStyleEl.textContent = '@page { size: A4 portrait; margin: 12mm; }';
        }

        document.getElementById('print-area').innerHTML = buildPrintHtml(inv, sizeMode);
        window.print();
      }

         ۱۱. پشتیبان‌گیری JSON و خروجی اکسل CSV فارسی با UTF-8 BOM
         ----------------------------------------------------------------------- */
      function exportJsonBackup() {
        try {
          const payload = {
            app: 'JavanCafeNetAccounting',
            version: 3,
            exportedAt: new Date().toISOString(),
            data: {
              services: state.services,
              invoices: state.invoices,
              nextInvoiceNumber: state.nextInvoiceNumber,
              debts: state.debts,
              customLogo: state.customLogo,
              settings: state.settings
            }
          };
          const blob = new Blob([JSON.stringify(payload, null, 2)], { type: 'application/json;charset=utf-8' });
          const url = URL.createObjectURL(blob);
          const a = document.createElement('a');
          a.href = url;
          a.download = `javan-cafenet-backup-${getTodayJalali().replace(/\//g, '-')}.json`;
          document.body.appendChild(a);
          a.click();
          a.remove();
          URL.revokeObjectURL(url);
          showToast('فایل پشتیبان JSON با موفقیت دانلود شد.', 'success');
        } catch (err) {
          showToast('خطا در ایجاد فایل پشتیبان.', 'error');
        }
      }

      function importJsonBackup(file) {
        const reader = new FileReader();
        reader.onload = (e) => {
          try {
            const parsed = JSON.parse(e.target.result);
            const d = parsed.data || parsed;
            if (!Array.isArray(d.services) || !Array.isArray(d.invoices)) {
              showToast('ساختار فایل پشتیبان نامعتبر است.', 'error');
              return;
            }
            openConfirm(
              'بازیابی فایل پشتیبان',
              `اطلاعات فعلی با ${d.services.length} خدمت، ${d.invoices.length} فاکتور${Array.isArray(d.debts) ? ' و ' + d.debts.length + ' نسیه' : ''} جایگزین شود؟`,
              () => {
                state.services = d.services;
                state.invoices = d.invoices.map(normalizeInvoice).filter(Boolean);
                if (d.nextInvoiceNumber && !isNaN(Number(d.nextInvoiceNumber))) {
                  state.nextInvoiceNumber = Number(d.nextInvoiceNumber);
                }
                if (Array.isArray(d.debts)) { state.debts = d.debts.filter(x => x && x.id); saveDebts(); }
                if (d.settings) state.settings = { ...DEFAULT_SETTINGS, ...d.settings };
                saveServices();
                saveInvoices();
                saveSettings();
                saveLogo(d.customLogo || null);
                resetInvoice(false);
                closeModal('settings-modal');
                showToast('اطلاعات با موفقیت بازیابی شد.', 'success');
              },
              'بازیابی اطلاعات'
            );
          } catch (err) {
            showToast('فایل انتخاب‌شده JSON معتبر نیست.', 'error');
          }
        };
        reader.readAsText(file);
      }

      function exportCsvExcel() {
        if (state.invoices.length === 0) {
          showToast('هیچ فاکتوری در تاریخچه برای خروجی اکسل وجود ندارد.', 'error');
          return;
        }
        const csvCell = (val) => `"${String(val ?? '').replace(/"/g, '""')}"`;
        const headers = ['شماره فاکتور', 'تاریخ شمسی', 'نام مشتری', 'شماره تماس', 'وضعیت پرداخت', 'تعداد اقلام', 'جمع کل (تومان)', 'تخفیف (تومان)', 'قابل پرداخت (تومان)', 'قابل پرداخت (ریال)'];
        const lines = [headers.map(csvCell).join(',')];

        state.invoices.forEach(inv => {
          lines.push([
            inv.invoiceNumber,
            inv.date,
            inv.customerName || 'مشتری حضوری',
            inv.customerPhone || '',
            inv.paymentStatus || 'پرداخت شده',
            inv.items ? inv.items.length : 0,
            inv.subtotal || 0,
            inv.discountAmount || 0,
            inv.finalToman || 0,
            inv.finalRial || 0
          ].map(csvCell).join(','));
        });

        // UTF-8 BOM برای نمایش بی‌نقص حروف فارسی در اکسل
        const bom = '\uFEFF';
        const blob = new Blob([bom + lines.join('\r\n')], { type: 'text/csv;charset=utf-8;' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = `javan-cafenet-invoices-${getTodayJalali().replace(/\//g, '-')}.csv`;
        document.body.appendChild(a);
        a.click();
        a.remove();
        URL.revokeObjectURL(url);
        showToast('فایل اکسل (CSV فارسی) دانلود شد.', 'success');
      }

"""Stable offline accounting content and optional local typefaces.

Financial data lives in the dedicated persistent WebEngine profile, not

from pathlib import Path
from zipfile import ZipFile, BadZipFile
from .storage import atomic_write

ASSETS = Path(__file__).resolve().parent.parent / 'assets'
FONT_NAMES = {
    'a rooznameh.ttf': 'Rooznameh.ttf',
    'far_nazanin.ttf': 'Nazanin.ttf',
    'vazirmatn_regular.ttf': 'Vazirmatn-Regular.ttf',
    'vazirmatn_medium.ttf': 'Vazirmatn-Medium.ttf',
    'vazirmatn_semibold.ttf': 'Vazirmatn-SemiBold.ttf',
    'vazirmatn_bold.ttf': 'Vazirmatn-Bold.ttf',
}
MAX_FONT_BYTES = 2 * 1024 * 1024


def write_if_changed(path, data):
    if not path.is_file() or path.read_bytes() != data:
        atomic_write(path, data)


def make_font_css(folder):
    rules = []
    def face(filename, family, weight):
        if (folder / filename).is_file():
            rules.append(f"@font-face {{ font-family: '{family}'; src: url('{filename}') format('truetype'); font-weight: {weight}; font-display: swap; }}")
            return True
        return False
    face('Vazirmatn.ttf', 'JavanVazir', '100 900')
    for name, weight in [('Regular', 400), ('Medium', 500), ('SemiBold', 600), ('Bold', 700)]:
        face(f'Vazirmatn-{name}.ttf', 'JavanVazirLocal', weight)
    face('Rooznameh.ttf', 'JavanHeadline', '400')
    face('Nazanin.ttf', 'JavanNazanin', '400')
    rules.append("""
html:root { --font-main: 'JavanVazirLocal', 'JavanVazir', 'Vazirmatn', Tahoma, sans-serif; }
.app-shell .brand-text h1, .print-doc .print-shop-name { font-family: 'JavanHeadline', var(--font-main); font-weight: 700; }
@media print {
  .print-doc:not(.print-80mm) .print-shop-info,
  .print-doc:not(.print-80mm) .print-words-box,
  .print-doc:not(.print-80mm) .print-signatures,
  .print-doc:not(.print-80mm) .print-footer-note {
    font-family: 'JavanNazanin', var(--font-main);
    font-size: 11pt;
  }
}
""")
    return '\n'.join(rules).encode('utf-8')


def prepare_accounting(data_folder, assets=ASSETS):
    # This file URL and directory name MUST remain constant across releases.
    root = Path(data_folder).resolve() / 'accounting'
    font_dir = root / 'fonts'
    font_dir.mkdir(parents=True, exist_ok=True)
    source = Path(assets) / 'accounting' / 'index.html'
    if not source.is_file():
        raise FileNotFoundError('فایل داخلی حسابداری در بسته برنامه یافت نشد.')
    write_if_changed(root / 'index.html', source.read_bytes())
    default_font = Path(assets) / 'fonts' / 'Vazirmatn.ttf'
    if default_font.is_file():
        write_if_changed(font_dir / 'Vazirmatn.ttf', default_font.read_bytes())
    license_file = Path(assets) / 'fonts' / 'OFL.txt'
    if license_file.is_file():
        write_if_changed(font_dir / 'OFL.txt', license_file.read_bytes())
    write_if_changed(font_dir / 'accounting.css', make_font_css(font_dir))
    return root / 'index.html'


def validate_font(data):
    if not 12 <= len(data) <= MAX_FONT_BYTES or data[:4] not in (b'\x00\x01\x00\x00', b'OTTO', b'true'):
        raise ValueError('فایل فونت معتبر نیست.')
    return data


def collect_fonts(paths):
    """Read only explicitly supported font basenames, not other ZIP contents."""
    result = {}
    for name in paths:
        path = Path(name)
        if path.suffix.lower() == '.zip':
            with ZipFile(path) as archive:
                for info in archive.infolist():
                    target = FONT_NAMES.get(info.filename.replace('\\', '/').rsplit('/', 1)[-1].casefold())
                    if not target or info.is_dir():
                        continue
                    if info.file_size > MAX_FONT_BYTES:
                        raise ValueError('حجم فونت بیش از حد مجاز است.')
                    with archive.open(info) as stream:
                        result[target] = validate_font(stream.read(MAX_FONT_BYTES + 1))
        else:
            target = FONT_NAMES.get(path.name.casefold())
            if target:
                with path.open('rb') as stream:
                    result[target] = validate_font(stream.read(MAX_FONT_BYTES + 1))
    return result


def import_fonts(data_folder, paths, assets=ASSETS):
    selected = collect_fonts(paths)
    if not selected:
        raise ValueError('هیچ‌یک از فونت‌های روزنامه، نازنین یا وزیرمتن مورد انتظار یافت نشد.')
    root = prepare_accounting(data_folder, assets).parent
    folder = root / 'fonts'
    for name, data in selected.items():
        write_if_changed(folder / name, data)
    write_if_changed(folder / 'accounting.css', make_font_css(folder))
    return tuple(sorted(selected))

class BrowserPage(QWebEnginePage):
    navigation_blocked = Signal(str)

    def acceptNavigationRequest(self, url, navigation_type, is_main_frame):
        if url.scheme().lower() not in ('http', 'https', 'about', 'data', 'blob'):
            self.navigation_blocked.emit(url.scheme())
            return False
        return super().acceptNavigationRequest(url, navigation_type, is_main_frame)


class LocalPolicy(QWebEngineUrlRequestInterceptor):
    def __init__(self, root, parent=None):
        super().__init__(parent)
        self.root = Path(root).resolve()

    def allows(self, url):
        if url.scheme().lower() in ('data', 'blob', 'about'):
            return True
        if url.isLocalFile() and not url.host():
            try:
                return Path(url.toLocalFile()).resolve().is_relative_to(self.root)
            except (OSError, ValueError, RuntimeError):
                return False
        return False

    def interceptRequest(self, info):
        if not self.allows(info.requestUrl()):
            info.block(True)


class LocalPage(BrowserPage):
    def __init__(self, profile, policy, parent):
        super().__init__(profile, parent)
        self.policy = policy

    def acceptNavigationRequest(self, url, navigation_type, is_main_frame):
        allowed = self.policy.allows(url)
        if not allowed:
            self.navigation_blocked.emit(url.scheme())
        return allowed


class BrowserSessions(QObject):
    """Profiles live for the application lifetime, longer than their pages."""
    def __init__(self, folder, download_handler):
        super().__init__(QApplication.instance())
        self.folder = Path(folder)
        self.download_handler = download_handler
        self.web = QWebEngineProfile(self)
        self.web.setPersistentCookiesPolicy(QWebEngineProfile.PersistentCookiesPolicy.NoPersistentCookies)
        self.web.downloadRequested.connect(download_handler)
        self.local_profiles = {}

    def local(self, root):
        root = Path(root).resolve()
        if root not in self.local_profiles:
            key = sha256(str(root).encode('utf-8')).hexdigest()[:32]
            profile = QWebEngineProfile('JavanLocal-' + key, self)
            profile.setPersistentStoragePath(str(self.folder / 'local-profiles' / key / 'storage'))
            profile.setCachePath(str(self.folder / 'local-profiles' / key / 'cache'))
            profile.setPersistentCookiesPolicy(QWebEngineProfile.PersistentCookiesPolicy.NoPersistentCookies)
            policy = LocalPolicy(root, profile)
            profile.setUrlRequestInterceptor(policy)
            profile.downloadRequested.connect(self.download_handler)
            self.local_profiles[root] = (profile, policy)
        return self.local_profiles[root]

"""Pure browser helpers; usable without Qt."""
from pathlib import Path
from urllib.parse import urlsplit

MIN_ZOOM, MAX_ZOOM, ZOOM_STEP = 50, 200, 10


def next_zoom(current: int, delta: int) -> int:
    return max(MIN_ZOOM, min(MAX_ZOOM, current + delta * ZOOM_STEP))


def safe_filename(name: str) -> str:
    # Strip either separator, Windows control/reserved characters and devices.
    name = name.replace('\\', '/').split('/')[-1]
    name = ''.join('_' if ord(c) < 32 or c in '<>:"/\\|?*' else c for c in name).strip(' .')
    stem = name.split('.')[0].upper()
    if stem in {'CON', 'PRN', 'AUX', 'NUL', *(f'COM{i}' for i in range(1, 10)), *(f'LPT{i}' for i in range(1, 10))}:
        name = '_' + name
    if len(name) > 180:
        suffix = Path(name).suffix[:20]
        name = name[:180-len(suffix)] + suffix
    return name or 'download'


def web_address(value: str) -> str:
    value = value.strip()
    if not value or any(c.isspace() or ord(c) < 32 for c in value) or '\\' in value:
        raise ValueError('آدرس اینترنتی معتبر نیست.')
    if '://' not in value:
        if ':' in value.split('/')[0]:
            raise ValueError('آدرس باید HTTP یا HTTPS باشد.')
        value = 'https://' + value
    parsed = urlsplit(value)
    if parsed.scheme not in ('https', 'http') or not parsed.hostname or parsed.username or parsed.password:
        raise ValueError('فقط آدرس HTTP یا HTTPS بدون اطلاعات ورود مجاز است.')
    _ = parsed.port
    return value


def origin_label(url: str) -> str:
    parsed = urlsplit(url)
    if parsed.scheme in ('http', 'https'):
        return parsed.scheme + '://' + parsed.netloc.rsplit('@', 1)[-1]
    return 'صفحه داخلی'

"""Validated link model and Persian search, independent of Qt."""
from dataclasses import dataclass
import unicodedata
from urllib.parse import urlsplit
from pathlib import Path, PureWindowsPath


@dataclass(frozen=True, slots=True)
class Link:
    id: str
    title: str
    url: str
    category: str
    description: str = ''
    kind: str = 'web'

    def __post_init__(self):
        for field, limit in ((self.id, 128), (self.title, 200), (self.category, 120),
                             (self.url, 8192), (self.description, 2000), (self.kind, 16)):
            if not isinstance(field, str) or len(field) > limit or '\x00' in field:
                raise ValueError('Invalid link field')
        if not all(value.strip() for value in (self.id, self.title, self.category, self.url)):
            raise ValueError('Link id, title, category and address are required')
        if self.kind == 'web':
            if any(ch.isspace() or ord(ch) < 32 for ch in self.url) or '\\' in self.url:
                raise ValueError('Invalid URL characters')
            parsed = urlsplit(self.url)
            if parsed.scheme not in ('http', 'https') or not parsed.hostname or parsed.username or parsed.password:
                raise ValueError('Expected an HTTP(S) URL without credentials')
            _ = parsed.port
        elif self.kind == 'html':
            path = Path(self.url)
            win = PureWindowsPath(self.url)
            if not (path.is_absolute() or win.is_absolute()) or self.url.startswith(('\\\\', '//')):
                raise ValueError('Expected an absolute local file path (not a network share)')
            if path.suffix.lower() not in ('.html', '.htm'):
                raise ValueError('Expected an HTML file')
        else:
            raise ValueError('Unknown link kind')

_TRANSLATION = str.maketrans({'ي': 'ی', 'ى': 'ی', 'ك': 'ک', 'ۀ': 'ه', 'ة': 'ه', '\u200c': ' ', '\u200d': ' ', '\u0640': '', **{a: str(i) for i, a in enumerate('۰۱۲۳۴۵۶۷۸۹')}, **{a: str(i) for i, a in enumerate('٠١٢٣٤٥٦٧٨٩')}})


def normalize(text: str) -> str:
    text = unicodedata.normalize('NFKC', text).translate(_TRANSLATION).casefold()
    return ' '.join(''.join(ch for ch in text if unicodedata.category(ch) not in ('Mn', 'Me', 'Cf')).split())


def filter_links(links: tuple[Link, ...], query: str) -> tuple[Link, ...]:
    terms = normalize(query).split()
    if not terms:
        return tuple(links)
    matches = []
    for link in links:
        field = normalize(' '.join((link.title, link.category, link.url, link.description)))
        compact = field.replace(' ', '')
        if all(term in field or term in compact for term in terms):
            matches.append(link)
    return tuple(matches)


def group_links(links: tuple[Link, ...]) -> dict[str, list[Link]]:
    groups: dict[str, list[Link]] = {}
    for link in links:
        groups.setdefault(link.category, []).append(link)
    return groups

"""Session-scoped download manager. Never opens downloaded files automatically."""
from pathlib import Path
from PySide6.QtCore import Qt, QStandardPaths, Signal, QTimer
from PySide6.QtWebEngineCore import QWebEngineDownloadRequest
from PySide6.QtWidgets import (QDialog, QFileDialog, QHBoxLayout, QLabel, QPushButton,
    QTableWidget, QTableWidgetItem, QVBoxLayout, QAbstractItemView, QHeaderView)
from .browser_utils import safe_filename
from .manager import warn

D = QWebEngineDownloadRequest.DownloadState


class DownloadManager(QDialog):
    changed = Signal()

    def __init__(self, parent):
        super().__init__(parent)
        self.setWindowTitle('مدیریت دانلودها')
        self.resize(820, 420)
        self.records = []
        self.refresh_timer = QTimer(self)
        self.refresh_timer.setSingleShot(True)
        self.refresh_timer.setInterval(120)
        self.refresh_timer.timeout.connect(self.refresh)
        layout = QVBoxLayout(self)
        self.summary = QLabel('دانلودهای این نشست')
        layout.addWidget(self.summary)
        self.table = QTableWidget(0, 4)
        self.table.setHorizontalHeaderLabels(['نام فایل', 'پیشرفت', 'وضعیت', 'مسیر ذخیره'])
        self.table.setSelectionBehavior(QAbstractItemView.SelectionBehavior.SelectRows)
        self.table.setSelectionMode(QAbstractItemView.SelectionMode.SingleSelection)
        self.table.setEditTriggers(QAbstractItemView.EditTrigger.NoEditTriggers)
        self.table.horizontalHeader().setSectionResizeMode(QHeaderView.ResizeMode.Interactive)
        self.table.horizontalHeader().setStretchLastSection(True)
        self.table.setColumnWidth(0, 190)
        self.table.setColumnWidth(1, 100)
        self.table.setColumnWidth(2, 145)
        layout.addWidget(self.table, 1)
        buttons = QHBoxLayout()
        for title, action in [('توقف موقت', self.pause), ('ادامه', self.resume), ('لغو', self.cancel), ('پاک کردن پایان‌یافته‌ها', self.clear_finished), ('بستن', self.hide)]:
            b = QPushButton(title)
            b.clicked.connect(action)
            buttons.addWidget(b)
        layout.addLayout(buttons)

    def receive(self, request):
        if request.isSavePageDownload() or request.state() != D.DownloadRequested:
            request.cancel()
            return
        folder = QStandardPaths.writableLocation(QStandardPaths.StandardLocation.DownloadLocation)
        suggested = str(Path(folder) / safe_filename(request.suggestedFileName()))
        name, _ = QFileDialog.getSaveFileName(self.parentWidget(), 'ذخیره فایل دانلودی', suggested, 'همه فایل‌ها (*)')
        if not name:
            request.cancel()
            return
        path = Path(name)
        if any(not rec['finished'] and rec['path'] == path for rec in self.records):
            request.cancel()
            warn(self.parentWidget(), 'دانلود دیگری در همین مسیر در حال انجام است.')
            return
        rec = {'request': request, 'path': path, 'finished': False, 'progress': '۰٪', 'status': 'در انتظار'}
        self.records.append(rec)
        request.setDownloadDirectory(str(path.parent))
        request.setDownloadFileName(path.name)
        request.receivedBytesChanged.connect(lambda: self.update_record(rec))
        request.totalBytesChanged.connect(lambda: self.update_record(rec))
        request.stateChanged.connect(lambda *_: self.update_record(rec))
        request.isPausedChanged.connect(lambda: self.update_record(rec))
        request.isFinishedChanged.connect(lambda: self.update_record(rec))
        request.destroyed.connect(lambda *_: self.request_destroyed(rec))
        request.accept()
        self.update_record(rec)
        self.show()
        self.raise_()

    def update_record(self, rec):
        r = rec['request']
        if r is None:
            return
        total, received = r.totalBytes(), r.receivedBytes()
        rec['progress'] = f'{min(100, int(received * 100 / total))}٪' if total > 0 else f'{received / 1048576:.1f} MB'
        state = r.state()
        rec['status'] = {D.DownloadRequested: 'در انتظار', D.DownloadInProgress: 'متوقف موقت' if r.isPaused() else 'در حال دانلود',
                         D.DownloadCompleted: 'کامل شد', D.DownloadCancelled: 'لغو شد', D.DownloadInterrupted: 'ناموفق: ' + r.interruptReasonString()}.get(state, 'نامشخص')
        rec['finished'] = r.isFinished()
        if state == D.DownloadCompleted:
            rec['progress'] = '۱۰۰٪'
        if not self.refresh_timer.isActive():
            self.refresh_timer.start()

    def request_destroyed(self, rec):
        rec['request'] = None
        if not rec['finished']:
            rec['status'] = 'پایان یافت'
            rec['finished'] = True
        if rec in self.records and not self.refresh_timer.isActive():
            self.refresh_timer.start()

    def refresh(self):
        row = self.table.currentRow()
        self.table.setRowCount(len(self.records))
        for i, rec in enumerate(self.records):
            for j, value in enumerate((rec['path'].name, rec['progress'], rec['status'], str(rec['path']))):
                self.table.setItem(i, j, QTableWidgetItem(value))
        if 0 <= row < len(self.records):
            self.table.selectRow(row)
        self.summary.setText(f'{len(self.records)} دانلود | {self.active_count()} فعال')
        self.changed.emit()

    def active_count(self):
        return sum(not rec['finished'] for rec in self.records)

    def selected(self):
        row = self.table.currentRow()
        if 0 <= row < len(self.records):
            rec = self.records[row]
            if not rec['finished']:
                return rec['request']
        return None

    def pause(self):
        r = self.selected()
        if r is not None and r.state() == D.DownloadInProgress:
            r.pause()

    def resume(self):
        r = self.selected()
        if r is not None and r.isPaused():
            r.resume()

    def cancel(self):
        r = self.selected()
        if r is not None:
            r.cancel()

    def cancel_all(self):
        for rec in list(self.records):
            if not rec['finished'] and rec['request'] is not None:
                rec['request'].cancel()

    def clear_finished(self):
        self.records = [rec for rec in self.records if not rec['finished']]
        self.refresh()
